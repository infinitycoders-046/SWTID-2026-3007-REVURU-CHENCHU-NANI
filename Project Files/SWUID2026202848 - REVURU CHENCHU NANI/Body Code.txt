<script>
(function () {

    // Set deadline
    const deadline = new Date("June 27, 2026 00:00:00").getTime();

    function updateCountdown() {

        const now = new Date().getTime();
        const distance = deadline - now;

        if (distance < 0) {
            clearInterval(timer);

            document.getElementById("days").innerHTML = "0";
            document.getElementById("hours").innerHTML = "0";
            document.getElementById("minutes").innerHTML = "0";
            document.getElementById("seconds").innerHTML = "0";

            document.getElementById("ended-message").innerHTML =
                "🎉 Deadline Reached!";
            return;
        }

        const days = Math.floor(distance / (1000 * 60 * 60 * 24));
        const hours = Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60));
        const minutes = Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60));
        const seconds = Math.floor((distance % (1000 * 60)) / 1000);

        document.getElementById("days").innerHTML = days;
        document.getElementById("hours").innerHTML = ("0" + hours).slice(-2);
        document.getElementById("minutes").innerHTML = ("0" + minutes).slice(-2);
        document.getElementById("seconds").innerHTML = ("0" + seconds).slice(-2);
    }

    updateCountdown();
    const timer = setInterval(updateCountdown, 1000);

})();
</script>

<script>

const sheetURL =
"https://docs.google.com/spreadsheets/d/e/2PACX-1vR3Kd70fRDrVLwOqQd5lToVjB8X7ivRZndW5oc3PrjMTBG1TGr0_Ruy2fJjOpsZxh43UBQk69ZX3BZh/pub?output=csv";

async function loadAnnouncements(){

    try{

        const response = await fetch(sheetURL);
        const csv = await response.text();

        const rows = csv
            .trim()
            .split("\n")
            .map(row => row.split(","));

        const dataRows = rows.slice(1).reverse();

        let html = '<div class="announcement-list">';

        dataRows.forEach((row,index)=>{

            const title = row[0] || "";
            const message = row[1] || "";
            const date = row[2] || "";

            html += `
                <div class="card">

                    <div class="card-header">

                        <div class="card-title">
                            ${title}
                        </div>

                        ${
                            index === 0
                            ? '<span class="badge">NEW</span>'
                            : ''
                        }

                    </div>

                    <div class="message">
                        ${message}
                    </div>

                    <div class="date">
                        📅 ${date}
                    </div>

                </div>
            `;
        });

        html += '</div>';

        document.getElementById("announcementBox").innerHTML =
            dataRows.length
            ? html
            : '<div class="loading">No announcements found</div>';

    }catch(error){

        document.getElementById("announcementBox").innerHTML =
            '<div class="error">Failed to load announcements</div>';

        console.error(error);
    }
}

loadAnnouncements();

setInterval(loadAnnouncements,30000);

</script></!doctype>
